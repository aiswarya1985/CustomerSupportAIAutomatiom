Step 3: Create an Artifact Registry Repository
Create a Docker repository to store your images. Use a simple name like support-copilot .

bash
gcloud artifacts repositories create support-copilot \
    --repository-format=docker \
    --location=us-central1 \
    --description="Docker repo for Support Copilot" \
    --project=x-victor-510811-b6
	
	
Step 4: Grant IAM Permissions to Cloud Build
Cloud Build uses a service account to push images and deploy to Cloud Run. Grant it the necessary roles using your Project Number (859126756364) .

bash
# Define variables for clarity
PROJECT=x-victor-510811-b6
CB_SA=859126756364-compute@developer.gserviceaccount.com

# Grant roles to the Cloud Build service account
for role in \
    roles/artifactregistry.writer \
    roles/run.admin \
    roles/iam.serviceAccountUser \
    roles/logging.logWriter
do
  gcloud projects add-iam-policy-binding $PROJECT \
    --member="serviceAccount:$CB_SA" \
    --role="$role"
done
